"""
Anonymous/lambda functions
    - when the function doesn't contain any name
    - main functionality - we can pass the argument and statements/expression in the SINGLE LINE
    syntax:
        lambda argument : expression

def add(a, b):
    return a + b

res = add(1, 2)
print(res)
"""

fun = lambda a, b : a + b
res = fun(2, 3)
print(res)
